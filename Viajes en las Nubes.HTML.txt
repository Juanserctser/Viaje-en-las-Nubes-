            // Axis-Aligned Bounding Box (AABB) Solid Wall Collision Helper
            function resolveSolidBoxCollision(box) {
                // Check if player bounding box overlaps box
                if (
                    arthur.x + arthur.radius > box.x &&
                    arthur.x - arthur.radius < box.x + box.w &&
                    arthur.y + arthur.radius > box.y &&
                    arthur.y - arthur.radius < box.y + box.h
                ) {
                    // Calculate penetration depths
                    const overlapLeft = (arthur.x + arthur.radius) - box.x;
                    const overlapRight = (box.x + box.w) - (arthur.x - arthur.radius);
                    const overlapTop = (arthur.y + arthur.radius) - box.y;
                    const overlapBottom = (box.y + box.h) - (arthur.y - arthur.radius);

                    // Find smallest overlap direction
                    const minOverlap = Math.min(overlapLeft, overlapRight, overlapTop, overlapBottom);

                    if (minOverlap === overlapLeft) {
                        arthur.x = box.x - arthur.radius;
                        arthur.vx = 0;
                    } else if (minOverlap === overlapRight) {
                        arthur.x = box.x + box.w + arthur.radius;
                        arthur.vx = 0;
                    } else if (minOverlap === overlapTop) {
                        arthur.y = box.y - arthur.radius;
                        arthur.vy = 0;
                        arthur.isGrounded = true;
                    } else if (minOverlap === overlapBottom) {
                        arthur.y = box.y + box.h + arthur.radius;
                        arthur.vy = Math.max(0, arthur.vy);
                    }
                }
            }

            function update() {
                if (isDead) return;

                // Handle Level 1 Completion Wave/Salute Animation
                if (inTransitionAnimation) {
                    transitionTimer++;
                    arthur.vx = 0;
                    arthur.vy = 0;
                    arthur.waveArmAngle = Math.sin(transitionTimer * 0.2) * 0.5;

                    if (transitionTimer > 150) { // ~2.5 seconds
                        setupLevel(2);
                    }
                    return;
                }

                // Update Moving Platforms & Disappearing Clouds
                for (let p of platforms) {
                    if (p.isMoving) {
                        p.x += p.speed * p.dir;
                        if (p.x > p.endX || p.x < p.startX) p.dir *= -1;
                    }
                    if (p.isDisappearing && p.timer > 0) {
                        p.timer++;
                        if (p.timer > 100) { // Disappear delay
                            p.active = false;
                        }
                        if (p.timer > 220) { // Respawn delay
                            p.timer = 0;
                            p.active = true;
                        }
                    }
                }

                // Controls physics
                if (keys.left) {
                    arthur.vx = -arthur.speed;
                    arthur.facingLeft = true;
                } else if (keys.right) {
                    arthur.vx = arthur.speed;
                    arthur.facingLeft = false;
                } else {
                    arthur.vx *= 0.8;
                }

                if (keys.jump && arthur.isGrounded) {
                    arthur.vy = arthur.jumpPower;
                    arthur.isGrounded = false;
                }

                arthur.vy += GRAVITY;

                // Apply Upward Wind Forces (Level 2 Windmill)
                for (let wz of windZones) {
                    if (
                        arthur.x > wz.x &&
                        arthur.x < wz.x + wz.w &&
                        arthur.y > wz.y &&
                        arthur.y < wz.y + wz.h
                    ) {
                        arthur.vy += wz.force;
                        if (arthur.vy < -9) arthur.vy = -9; // Cap vertical wind velocity
                    }
                }

                arthur.x += arthur.vx;
                arthur.y += arthur.vy;

                if (arthur.x - arthur.radius < 0) arthur.x = arthur.radius;

                // Fall into void death
                if (arthur.y > canvas.height + 100) {
                    showDeathScreen();
                    return;
                }

                // Platform Collisions
                arthur.isGrounded = false;
                for (let p of platforms) {
                    if (p.isDisappearing && !p.active) continue;

                    if (
                        arthur.x + arthur.radius > p.x &&
                        arthur.x - arthur.radius < p.x + p.w &&
                        arthur.y + arthur.radius >= p.y &&
                        arthur.y + arthur.radius <= p.y + p.h + arthur.vy &&
                        arthur.vy >= 0
                    ) {
                        arthur.isGrounded = true;
                        arthur.vy = 0;
                        arthur.y = p.y - arthur.radius;

                        if (p.isMoving) arthur.x += p.speed * p.dir;
                        if (p.isDisappearing && p.timer === 0) p.timer = 1; // Trigger disappearing timer
                    }
                }

                // Solid collisions for barrier blocks above door
                for (let block of doorBlockStack) {
                    resolveSolidBoxCollision(block);
                }

                // Solid collision for Door when closed
                if (door && !door.isOpen) {
                    resolveSolidBoxCollision(door);
                }

                // Pressure Buttons (Level 2)
                for (let btn of pressureButtons) {
                    if (
                        arthur.x + arthur.radius > btn.x &&
                        arthur.x - arthur.radius < btn.x + btn.w &&
                        arthur.y + arthur.radius >= btn.y &&
                        arthur.y - arthur.radius <= btn.y + btn.h
                    ) {
                        btn.pressed = true;
                    } else {
                        btn.pressed = false;
                    }
                }

                // Level 1 Key Pick
                if (keyItem && !keyItem.collected) {
                    const distToKey = Math.hypot(arthur.x - keyItem.x, arthur.y - keyItem.y);
                    if (distToKey < 35) {
                        keyItem.collected = true;
                        arthur.hasKey = true;
                    }
                }

                // Door & Puzzle Interaction Trigger
                const distToDoor = Math.hypot(arthur.x - (door.x + door.w / 2), arthur.y - (door.y + door.h / 2));
                if (distToDoor < 95 && !door.isOpen) {
                    btnAction.classList.remove('hidden');
                    if (keys.action) {
                        keys.action = false;
                        if (door.isPuzzle) {
                            puzzleModal.classList.remove('hidden');
                        } else if (arthur.hasKey) {
                            door.isOpen = true;
                            btnAction.classList.add('hidden');
                        }
                    }
                } else {
                    btnAction.classList.add('hidden');
                }

                // Goal Portal Collision & Level Transition
                const distToGoal = Math.hypot(arthur.x - goal.x, arthur.y - goal.y);
                if (distToGoal < goal.radius + arthur.radius) {
                    if (currentLevel === 1 && !inTransitionAnimation) {
                        // Start Arthur Salute/Wave Animation
                        inTransitionAnimation = true;
                        transitionTimer = 0;
                        speechBubbleText = "¡Hola! ¡Al Nivel 2! 👋";
                    } else if (currentLevel === 2) {
                        victoryModal.classList.remove('hidden');
                    }
                }

                // Camera follow
                cameraX = arthur.x - canvas.width * 0.35;
                if (cameraX < 0) cameraX = 0;
                if (cameraX > worldWidth - canvas.width) cameraX = worldWidth - canvas.width;

                // Tutorial Texts
                if (currentLevel === 1) {
                    if (arthur.x < 350) tutorialText.innerHTML = "<strong>Tutorial:</strong> Usa ◀ ▶ para mover a Arthur.";
                    else if (!keyItem.collected) tutorialText.innerHTML = "<strong>Misión:</strong> Salta las nubes y recoge la 🔑 LLAVE.";
                    else if (!door.isOpen) tutorialText.innerHTML = "<strong>Misión:</strong> Acércate a la puerta y pulsa ⚙️ para abrirla.";
                    else tutorialText.innerHTML = "<strong>Meta:</strong> ¡Cruza el portal ⭐ para pasar de nivel!";
                } else {
                    if (arthur.x < 500) tutorialText.innerHTML = "<strong>Nivel 2:</strong> Salta sobre las nubes flotantes.";
                    else if (arthur.x < 1100) tutorialText.innerHTML = "<strong>Molino de Viento:</strong> ¡Entra en la corriente de aire para elevarte! 💨";
                    else if (arthur.x < 1700) tutorialText.innerHTML = "<strong>Cuidado:</strong> ¡Las nubes oscuras se desvanecen!";
                    else if (!door.isOpen) tutorialText.innerHTML = "<strong>Acertijo:</strong> Pulsa 🧩 en la puerta para resolver el enigma del Sol, Luna y Estrella.";
                    else tutorialText.innerHTML = "<strong>Meta:</strong> ¡El camino está despejado! Ve al Portal ⭐";
                }
            }

            function render() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);

                ctx.save();
                ctx.translate(-cameraX, 0);

                // Background Sky Clouds
                ctx.fillStyle = "rgba(255, 255, 255, 0.45)";
                for (let c of bgClouds) {
                    ctx.beginPath();
                    ctx.arc(c.x, c.y, 25 * c.scale, 0, Math.PI * 2);
                    ctx.arc(c.x + 15 * c.scale, c.y - 10 * c.scale, 20 * c.scale, 0, Math.PI * 2);
                    ctx.arc(c.x + 30 * c.scale, c.y, 25 * c.scale, 0, Math.PI * 2);
                    ctx.fill();
                }

                // Windmill & Air Currents (Level 2)
                for (let wm of windmills) {
                    wm.rotation += 0.03;
                    // Windmill Stand
                    ctx.fillStyle = "#94a3b8";
                    ctx.beginPath();
                    ctx.moveTo(wm.x - 6, wm.y);
                    ctx.lineTo(wm.x + 6, wm.y);
                    ctx.lineTo(wm.x + 2, wm.y - 70);
                    ctx.lineTo(wm.x - 2, wm.y - 70);
                    ctx.fill();

                    // Blades
                    ctx.save();
                    ctx.translate(wm.x, wm.y - 70);
                    ctx.rotate(wm.rotation);
                    ctx.fillStyle = "#f8fafc";
                    for (let b = 0; b < 3; b++) {
                        ctx.rotate((Math.PI * 2) / 3);
                        ctx.beginPath();
                        ctx.roundRect(-4, 0, 8, 45, 4);
                        ctx.fill();
                    }
                    ctx.restore();
                }

                // Wind Zones Visual Streams
                for (let wz of windZones) {
                    ctx.fillStyle = "rgba(255, 255, 255, 0.2)";
                    ctx.fillRect(wz.x, wz.y, wz.w, wz.h);

                    ctx.fillStyle = "rgba(255, 255, 255, 0.65)";
                    for (let i = 0; i < 6; i++) {
                        let lineY = (Date.now() / 3 + i * 50) % wz.h;
                        ctx.fillRect(wz.x + 12 + i * 12, wz.y + wz.h - lineY, 3, 22);
                    }
                }

                // Platforms Render
                for (let p of platforms) {
                    if (p.isDisappearing && !p.active) continue;

                    let alpha = 1;
                    if (p.isDisappearing && p.timer > 0) {
                        alpha = Math.max(0.2, 1 - p.timer / 100);
                    }

                    ctx.globalAlpha = alpha;
                    ctx.fillStyle = p.isDisappearing ? "#93c5fd" : (p.isMoving ? "#e0f2fe" : "#ffffff");
                    ctx.shadowColor = "rgba(14, 165, 233, 0.2)";
                    ctx.shadowBlur = 12;
                    ctx.shadowOffsetY = 6;

                    ctx.beginPath();
                    ctx.roundRect(p.x, p.y, p.w, p.h, 20);
                    ctx.fill();

                    ctx.shadowColor = 'transparent';
                    ctx.fillStyle = p.isDisappearing ? "#60a5fa" : "#f0f9ff";
                    for (let xBump = p.x + 15; xBump < p.x + p.w - 10; xBump += 25) {
                        ctx.beginPath();
                        ctx.arc(xBump, p.y, 9, 0, Math.PI * 2);
                        ctx.fill();
                    }
                    ctx.globalAlpha = 1;
                }

                // Barrier Blocks Stack Above Door (50 Blocks Height)
                for (let block of doorBlockStack) {
                    ctx.fillStyle = "#64748b";
                    ctx.strokeStyle = "#334155";
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    ctx.roundRect(block.x, block.y, block.w, block.h, 6);
                    ctx.fill();
                    ctx.stroke();

                    // Block texture pattern
                    ctx.fillStyle = "#475569";
                    ctx.fillRect(block.x + 6, block.y + 6, block.w - 12, block.h - 12);
                }

                // Pressure Buttons (Level 2)
                for (let btn of pressureButtons) {
                    ctx.fillStyle = btn.pressed ? "#22c55e" : "#ef4444";
                    const btnH = btn.pressed ? 5 : btn.h;
                    ctx.fillRect(btn.x, btn.y + (btn.h - btnH), btn.w, btnH);
                }

                // Level 1 Key Item
                if (keyItem && !keyItem.collected) {
                    ctx.shadowColor = "#f59e0b";
                    ctx.shadowBlur = 15;
                    ctx.fillStyle = "#fbbf24";
                    ctx.font = "28px sans-serif";
                    ctx.fillText("🔑", keyItem.x - 14, keyItem.y + 10 + Math.sin(Date.now() / 200) * 4);
                    ctx.shadowColor = 'transparent';
                }

                // Door Render
                if (door) {
                    ctx.fillStyle = door.isOpen ? "#22c55e" : (door.isPuzzle ? "#0284c7" : "#854d0e");
                    ctx.beginPath();
                    ctx.roundRect(door.x, door.y, door.w, door.h, [12, 12, 0, 0]);
                    ctx.fill();

                    ctx.fillStyle = "#ffffff";
                    ctx.font = "22px sans-serif";
                    ctx.fillText(door.isOpen ? "🔓" : (door.isPuzzle ? "🧩" : "🔒"), door.x + 18, door.y + 42);
                }

                // Goal Portal Render
                ctx.shadowColor = "#facc15";
                ctx.shadowBlur = 22;
                ctx.fillStyle = "#fef08a";
                ctx.beginPath();
                ctx.arc(goal.x, goal.y, goal.radius, 0, Math.PI * 2);
                ctx.fill();
                ctx.font = "36px sans-serif";
                ctx.fillText("⭐", goal.x - 18, goal.y + 12);
                ctx.shadowColor = 'transparent';

                // Arthur Player Character
                if (!isDead) {
                    ctx.shadowColor = "rgba(0, 0, 0, 0.3)";
                    ctx.shadowBlur = 8;
                    ctx.shadowOffsetY = 4;

                    // Arthur Body
                    ctx.fillStyle = "#0f172a";
                    ctx.beginPath();
                    ctx.arc(arthur.x, arthur.y, arthur.radius, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.shadowColor = 'transparent';

                    // Cute White Eyes
                    ctx.fillStyle = "#ffffff";
                    let eyeDir = arthur.facingLeft ? -4 : 4;
                    if (inTransitionAnimation) eyeDir = 0; // Look directly at screen

                    ctx.beginPath();
                    ctx.arc(arthur.x + eyeDir - 3.5, arthur.y - 3, 3.8, 0, Math.PI * 2);
                    ctx.arc(arthur.x + eyeDir + 3.5, arthur.y - 3, 3.8, 0, Math.PI * 2);
                    ctx.fill();

                    // Pupils
                    ctx.fillStyle = "#000000";
                    ctx.beginPath();
                    ctx.arc(arthur.x + eyeDir - 3, arthur.y - 3, 1.6, 0, Math.PI * 2);
                    ctx.arc(arthur.x + eyeDir + 4, arthur.y - 3, 1.6, 0, Math.PI * 2);
                    ctx.fill();

                    // Waving Arm Animation when level completed
                    if (inTransitionAnimation) {
                        ctx.strokeStyle = "#0f172a";
                        ctx.lineWidth = 4;
                        ctx.lineCap = "round";

                        // Waving hand
                        ctx.save();
                        ctx.translate(arthur.x + 12, arthur.y - 2);
                        ctx.rotate(-0.5 + arthur.waveArmAngle);
                        ctx.beginPath();
                        ctx.moveTo(0, 0);
                        ctx.lineTo(12, -8);
                        ctx.stroke();
                        ctx.restore();

                        // Cute Speech Bubble
                        ctx.fillStyle = "#ffffff";
                        ctx.strokeStyle = "#0284c7";
                        ctx.lineWidth = 2;
                        ctx.beginPath();
                        ctx.roundRect(arthur.x - 65, arthur.y - 70, 130, 36, 12);
                        ctx.fill();
                        ctx.stroke();

                        // Bubble pointer
                        ctx.beginPath();
                        ctx.moveTo(arthur.x - 5, arthur.y - 34);
                        ctx.lineTo(arthur.x + 5, arthur.y - 34);
                        ctx.lineTo(arthur.x, arthur.y - 24);
                        ctx.fill();

                        ctx.fillStyle = "#0369a1";
                        ctx.font = "bold 12px 'Segoe UI', sans-serif";
                        ctx.textAlign = "center";
                        ctx.fillText(speechBubbleText, arthur.x, arthur.y - 47);
                        ctx.textAlign = "left";
                    }
                }

                ctx.restore();
            }

            // Main Game Loop
            function gameLoop() {
 update();
                render();
                requestAnimationFrame(gameLoop);
            }

            // Start Game on Level 1
            setupLevel(1);
            gameLoop();
        };
    </script>
